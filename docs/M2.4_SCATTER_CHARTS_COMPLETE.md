## Technical Implementation Details

### ScatterDataPoint Design

Initially attempted to inherit from `DataPoint`, but this failed because `DataPoint` is a `readonly struct` (sealed type). Solution: implement `IDataPoint` directly:

```csharp
public class ScatterDataPoint : Data.IDataPoint
{
    public ScatterDataPoint(double x, double y) { X = x; Y = y; }
    public ScatterDataPoint(double x, double y, double size) {
        X = x; Y = y; Size = size;
    }
    public ScatterDataPoint(double x, double y, double size, double colorValue) {
        X = x; Y = y; Size = size; ColorValue = colorValue;
    }

    public double X { get; }
    public double Y { get; }
    public double Size { get; }
    public double ColorValue { get; }
}
```

### Variable Sizing Algorithm

```csharp
// Calculate min/max size values from data
foreach (var point in series)
{
    if (point is ScatterDataPoint scp)
    {
        minSize = Math.Min(minSize, scp.Size);
        maxSize = Math.Max(maxSize, scp.Size);
    }
}

// Handle edge case
if (minSize == maxSize) maxSize = minSize + 1;

// Map data size to pixel size
var normalizedSize = (scp.Size - minSize) / (maxSize - minSize);
markerSize = MinMarkerSize +
    (float)(normalizedSize * (MaxMarkerSize - MinMarkerSize));
```

### Color Mapping Algorithm

Uses linear interpolation between colors in a scale:

```csharp
private SKColor MapValueToColor(double value, double minValue, double maxValue,
    ScatterSeriesStyle style)
{
    // Normalize value to 0-1
    var normalizedValue = (value - minValue) / (maxValue - minValue);

    // Default 5-color scale
    var colorScale = style.ColorScale ?? new[] {
        new SKColor(0, 0, 255),    // Blue (low)
        new SKColor(0, 255, 255),  // Cyan
        new SKColor(0, 255, 0),    // Green
        new SKColor(255, 255, 0),  // Yellow
        new SKColor(255, 0, 0)     // Red (high)
    };

    // Scale to color array indices
    var scaledValue = normalizedValue * (colorScale.Length - 1);
    var index = (int)Math.Floor(scaledValue);
    var t = scaledValue - index;

    // Interpolate RGB components
    var color1 = colorScale[index];
    var color2 = colorScale[index + 1];
    var r = (byte)(color1.Red + (color2.Red - color1.Red) * t);
    var g = (byte)(color1.Green + (color2.Green - color1.Green) * t);
    var b = (byte)(color1.Blue + (color2.Blue - color1.Blue) * t);

    return new SKColor(r, g, b);
}
```

## Usage Example

```csharp
var chart = new ScatterChart();

// Enable variable sizing and color mapping
chart.DefaultStyle.UseVariableSizes = true;
chart.DefaultStyle.MinMarkerSize = 6f;
chart.DefaultStyle.MaxMarkerSize = 18f;
chart.DefaultStyle.UseColorMapping = true;
chart.DefaultStyle.MarkerShape = MarkerShape.Circle;
chart.DefaultStyle.BorderColor = SKColors.Black;
chart.DefaultStyle.BorderWidth = 1f;

// Create scatter data with size and color values
var series = new DataSeries<IDataPoint>(new IDataPoint[]
{
    new ScatterDataPoint(0, 10, 5, 10),   // x, y, size, colorValue
    new ScatterDataPoint(1, 20, 15, 50),
    new ScatterDataPoint(2, 15, 25, 90),
});

chart.Series.Add(series);

// Render
chart.Render(canvas, width, height);
```
