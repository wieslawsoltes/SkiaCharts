```csharp
private bool HandleMouseWheel(MouseEvent mouseEvent)
{
    if (!MouseWheelZoomEnabled)
        return false;

    var factor = mouseEvent.WheelDelta > 0 ? MouseWheelZoomFactor : 1.0f / MouseWheelZoomFactor;
    _viewport.ZoomBy(factor, mouseEvent.Position);

    return true;
```

  - Automatic pan adjustment

**Key Implementation**:
```csharp
public void ZoomToFit(float margin = 0.1f)
{
    var zoomX = viewWidth / dataWidth;
    var zoomY = viewHeight / dataHeight;
    var zoom = Math.Min(zoomX, zoomY);

    zoom *= (1.0f - margin); // Apply margin
    zoom = Math.Clamp(zoom, MinZoom, MaxZoom);

    _zoom = zoom;

    // Center the data in the view
    var visible = VisibleDataRect;
    _pan.X = _dataBounds.Left - (visible.Width - dataWidth) / 2;
    _pan.Y = _dataBounds.Top - (visible.Height - dataHeight) / 2;
}
```

```csharp
public void AnimateZoomAndPan(float targetZoom, SKPoint targetPan,
    TimeSpan? duration = null, EasingFunction? easing = null)
{
    StartAnimation(new AnimationState
    {
        StartZoom = _viewport.Zoom,
        TargetZoom = targetZoom,
        StartPan = _viewport.Pan,
        TargetPan = targetPan,
        Duration = duration ?? DefaultDuration,
        Easing = easing ?? DefaultEasing
    });
}

public bool Update()
{
    var progress = Math.Min(1.0, elapsed.TotalMilliseconds / _currentAnimation.Duration.TotalMilliseconds);
    var easedProgress = ApplyEasing(progress, _currentAnimation.Easing);

    // Interpolate zoom and pan
    _viewport.Zoom = Lerp(startZoom, targetZoom, easedProgress);
    _viewport.Pan = Lerp(startPan, targetPan, easedProgress);
}
```

**Key Implementation**:
```csharp
public float Zoom
{
    get => _zoom;
    set
    {
        var newZoom = Math.Clamp(value, MinZoom, MaxZoom);
        if (Math.Abs(_zoom - newZoom) > 0.0001f)
        {
            _zoom = newZoom;
            OnTransformChanged();
        }
    }
}

public float MinZoom { get; set; } = 0.1f;
public float MaxZoom { get; set; } = 100.0f;
```

## Architecture Highlights

### Navigation Flow
```
User Input (Mouse/Touch/Keyboard)
    ↓
NavigationBehavior (High-level handler)
    ↓
Viewport (Core transformation logic)
    ↓
TransformChanged Event
    ↓
Chart Re-render
```

### Animation Flow
```
NavigationBehavior.AnimateTransitions = true
    ↓
ViewportAnimation.Animate*()
    ↓
Frame Loop: Update()
    ↓
Viewport.Zoom/Pan (interpolated values)
    ↓
TransformChanged Event
    ↓
Chart Re-render
```

### Coordinate System
```
Data Space (raw data coordinates)
    ↓ ZoomBy(factor) + PanBy(delta)
Visible Data Rect (zoomed & panned view)
    ↓ DataToScreen() transform
Screen Space (pixel coordinates)
```
