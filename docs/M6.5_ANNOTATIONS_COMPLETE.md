```csharp
var customAnnotation = new CustomAnnotation
{
    Name = "Custom Shape",
    RenderFunction = (canvas, chartBounds, viewport) =>
    {
        // Convert data coordinates to screen coordinates
        var point1 = viewport.DataToScreen(new SKPoint(100, 50));
        var point2 = viewport.DataToScreen(new SKPoint(150, 75));

        // Custom rendering logic
        using var paint = new SKPaint
        {
            Color = SKColors.Purple,
            Style = SKPaintStyle.Stroke,
            StrokeWidth = 3
        };

        canvas.DrawLine(point1, point2, paint);
        canvas.DrawCircle(point1, 5, paint);
        canvas.DrawCircle(point2, 5, paint);
    },
    HitTestFunction = (position, viewport) =>
    {
        // Custom hit testing logic
        var point1 = viewport.DataToScreen(new SKPoint(100, 50));
        var distance = Math.Sqrt(
            Math.Pow(position.X - point1.X, 2) +
            Math.Pow(position.Y - point1.Y, 2));
        return distance <= 10;
    }
};
```

## Architecture Highlights

### Annotation System Flow
```
Chart → AnnotationManager
    ↓
AddAnnotation() → IAnnotation
    ↓
Render() → Canvas + Viewport
    ↓
    ├─ PointAnnotation (marker + label)
    ├─ RangeAnnotation (vertical band)
    ├─ ThresholdAnnotation (horizontal line)
    └─ CustomAnnotation (user-defined)
    ↓
HitTest() → Detect Clicks
    ↓
Events → Application Logic
```

### Coordinate Transformation
```
Data Space (X, Y) → Viewport.DataToScreen() → Screen Space (pixels)
                                           ↓
                                    Render on Canvas
```

### Event-Driven Architecture
All annotation operations use event-driven patterns:
- `AnnotationAdded` event - Track when annotations are added
- `AnnotationRemoved` event - Track when annotations are removed
- `AnnotationClicked` event - React to user interactions

## Technical Specifications

### Performance Characteristics
- **Add Annotation**: O(1) - Direct list append
- **Remove Annotation**: O(n) for n annotations - linear search
- **Hit Test**: O(n) for n annotations - reverse iteration
- **Render**: O(n) for n annotations - single pass
- **Clear**: O(1) - List clear operation

### Memory Footprint
- AnnotationManager: ~150 bytes + O(n) for n annotations
- PointAnnotation: ~180 bytes per instance
- RangeAnnotation: ~150 bytes per instance
- ThresholdAnnotation: ~160 bytes per instance
- CustomAnnotation: ~120 bytes + function closures
- Total per chart with 10 annotations: ~1,650 bytes

### Thread Safety
- Not thread-safe by design (UI thread only)
- Event handlers execute synchronously
- No internal locking required

## Design Patterns Used

✅ **Manager Pattern**: AnnotationManager controls all annotations
✅ **Strategy Pattern**: Pluggable annotation types via IAnnotation
✅ **Observer Pattern**: Event-based notifications
✅ **Template Method**: AnnotationBase provides base implementation
✅ **Null Object Pattern**: Empty render/hit test functions for custom annotations
✅ **Visitor Pattern**: Hit testing traverses annotations

### Custom Annotation
```csharp
// Custom arrow annotation
var arrow = new CustomAnnotation
{
    Name = "Trend Arrow",
    RenderFunction = (canvas, chartBounds, viewport) =>
    {
        var start = viewport.DataToScreen(new SKPoint(100, 50));
        var end = viewport.DataToScreen(new SKPoint(150, 75));

        using var paint = new SKPaint
        {
            Color = SKColors.Purple,
            Style = SKPaintStyle.Stroke,
            StrokeWidth = 3,
            IsAntialias = true
        };

        // Draw arrow line
        canvas.DrawLine(start, end, paint);

        // Draw arrowhead
        var angle = Math.Atan2(end.Y - start.Y, end.X - start.X);
        var arrowSize = 10;

        var p1 = new SKPoint(
            end.X - arrowSize * (float)Math.Cos(angle - Math.PI / 6),
            end.Y - arrowSize * (float)Math.Sin(angle - Math.PI / 6));
        var p2 = new SKPoint(
            end.X - arrowSize * (float)Math.Cos(angle + Math.PI / 6),
            end.Y - arrowSize * (float)Math.Sin(angle + Math.PI / 6));

        canvas.DrawLine(end, p1, paint);
        canvas.DrawLine(end, p2, paint);
    },
    HitTestFunction = (position, viewport) =>
    {
        // Hit test on arrow line
        var start = viewport.DataToScreen(new SKPoint(100, 50));
        var end = viewport.DataToScreen(new SKPoint(150, 75));

        // Distance from point to line segment
        var dx = end.X - start.X;
        var dy = end.Y - start.Y;
        var lengthSquared = dx * dx + dy * dy;

        if (lengthSquared == 0)
            return false;

        var t = Math.Max(0, Math.Min(1,
            ((position.X - start.X) * dx + (position.Y - start.Y) * dy) / lengthSquared));

        var projX = start.X + t * dx;
        var projY = start.Y + t * dy;

        var distance = Math.Sqrt(
            (position.X - projX) * (position.X - projX) +
            (position.Y - projY) * (position.Y - projY));

        return distance <= 5; // 5px tolerance
    }
};

annotationMgr.AddAnnotation(arrow);
```
